        function createPetal() {
            const petal = document.createElement('div');
            petal.className = 'petal';
            petal.style.setProperty('--left', Math.random() * 100 + 'vw');
            petal.style.width = Math.random() * 15 + 10 + 'px';
            petal.style.height = petal.style.width;
            petal.style.animationDuration = Math.random() * 5 + 5 + 's';
            document.getElementById('petals-container').appendChild(petal);
            setTimeout(() => petal.remove(), 10000);
        }

        function openLetter() {
            document.getElementById('envelope-overlay').style.transform = 'translateY(-100%)';
            const content = document.getElementById('main-content');
            content.classList.remove('opacity-0');
            setInterval(createPetal, 500);
            
            window.addEventListener('scroll', () => {
                const sig = document.getElementById('signature-trigger');
                if (sig.getBoundingClientRect().top < window.innerHeight) {
                    document.getElementById('sig-container').classList.remove('opacity-0');
                    document.getElementById('sig-container').classList.add('animate-sig');
                }
            });
